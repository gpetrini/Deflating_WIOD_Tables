* Global constraints

- Package name =growthdecomp=; license GPL-3 (ADR-0008).
- Input contract is a NIOT object: named list by ISO; each country holds per-year =Ft=, =Fn=, =Fm= (data frames with columns =HFCE, GGFC, GFCF, EXPO, INVNT=), =lista_Am=, =lista_Z= (matrices). Slots =Recurso=, =CIn=, =CIm=, =lista_An=, =lista_A= are not read (ADR-0001).
- Exported computation functions return objects and perform no file I/O; one impure wrapper =report_all()= writes (ADR-0004).
- No global reads; =countries= and =years= derive from the NIOT, groups are an explicit argument (ADR-0004).
- Native pipe =|>=; =ggplot2= for graphics; tables via =gt= or =knitr::kable(booktabs = TRUE)=.
- Viz, table, data-access, and test packages are Suggests, guarded with =requireNamespace()= (spec, dependency split).
- No API key is committed or shipped; keyed fetchers read =Sys.getenv()= (ADR-0007).
- Data-gated tasks (real sample, identity tests, built vignette) require a WIOD-derived =niot_wiod=; until it exists their verification is structural, mirroring the accepted diagnosis-plan pattern.

## A minimal, accounting-shaped NIOT for contract tests. No external data.
test_niot <- function() {
  yrs <- as.character(2001:2003)
  cols <- c("HFCE", "GGFC", "GFCF", "EXPO", "INVNT")

  fd <- function(scale) {
    setNames(as.data.frame(matrix(scale * c(10, 6, 4, 8, 2,
                                            5, 3, 2, 4, 1),
                                  nrow = 2, byrow = TRUE)), cols)
  }
  mat <- function(v) matrix(v, nrow = 2)

  per_year <- function(k) list(
    Ft = fd(1.00 * k), Fn = fd(0.70 * k), Fm = fd(0.30 * k),
    lista_Am = mat(c(0.10, 0.05, 0.04, 0.08)),
    lista_Z  = mat(c(1.12, 0.06, 0.05, 1.10))
  )

  slots <- c("Ft", "Fn", "Fm", "lista_Am", "lista_Z")
  country <- setNames(vector("list", length(slots)), slots)
  for (s in slots) country[[s]] <- setNames(lapply(seq_along(yrs),
                                    function(i) per_year(1 + 0.03 * i)[[s]]), yrs)
  list(AAA = country)
}

#+begin_src R
#' @export
fetch_fred <- function(series_id) {
  key <- Sys.getenv("FRED_API_KEY")
  if (!nzchar(key)) stop("Set FRED_API_KEY in ~/.Renviron; get a free key at https://fred.stlouisfed.org/docs/api/api_key.html")
  if (!requireNamespace("fredr", quietly = TRUE)) stop("Install 'fredr'.")
  fredr::fredr_set_key(key)
  fredr::fredr(series_id) |>
    dplyr::transmute(ISO = NA_character_, Year = as.integer(format(date, "%Y")), value)
}

#' @export
fetch_wb <- function(indicator, countries) {
  if (!requireNamespace("WDI", quietly = TRUE)) stop("Install 'WDI'.")
  WDI::WDI(country = countries, indicator = indicator) |>
    dplyr::transmute(ISO = iso3c, Year = year, value = .data[[indicator]])
}
#+end_src
