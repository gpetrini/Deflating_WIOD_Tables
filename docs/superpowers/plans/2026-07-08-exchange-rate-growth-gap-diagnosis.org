## Sources the pipeline once and caches the prepared results in an env.
.io_cache <- new.env()

load_io <- function() {
  if (is.null(.io_cache$results)) {
    source("global_variables.R")      # loads NIOTs, defines WIOD_countries, years
    source("support_functions.R")
    .io_cache$results <- prepare_data()
  }
  .io_cache$results
}

get_gIO <- function(results = load_io()) {
  map(names(results), function(iso) {
    lvl <- results[[iso]][["GDP"]][, "GDP"]
    yrs <- format(index(lvl), "%Y")
    g <- diff(log(as.numeric(lvl)))
    tibble(ISO = iso, Year = as.integer(yrs[-1]), gY_io = g)
  }) |>
    list_rbind()
}

#+begin_src R
get_gOff <- function(
  path = "../inputs/world_bank/GDP_constant/API_NY.GDP.MKTP.KD_DS2_en_csv_v2.csv"
) {
  raw <- read_csv(path, skip = 4, show_col_types = FALSE)
  raw |>
    rename(ISO = `Country Code`) |>
    select(ISO, matches("^[0-9]{4}$")) |>
    pivot_longer(-ISO, names_to = "Year", values_to = "gdp_kd") |>
    mutate(Year = as.integer(Year)) |>
    filter(!is.na(gdp_kd)) |>
    arrange(ISO, Year) |>
    group_by(ISO) |>
    mutate(gY_off = c(NA, diff(log(gdp_kd)))) |>
    ungroup() |>
    filter(!is.na(gY_off)) |>
    select(ISO, Year, gY_off)
}

#+begin_src R
get_xr_infl <- function() {
  source("tmp.R")                       # defines get_Erate, get_deflator, get_common
  e <- get_Erate()  |> transmute(ISO = Code, Year = as.integer(Year), e = Value)
  P <- get_deflator() |> transmute(ISO = Code, Year = as.integer(Year), P = Value)

  e |>
    inner_join(P, by = c("ISO", "Year")) |>
    arrange(ISO, Year) |>
    group_by(ISO) |>
    mutate(dln_e = c(NA, diff(log(e))),
           pi    = c(NA, diff(log(P)))) |>
    ungroup() |>
    filter(!is.na(dln_e), !is.na(pi)) |>
    select(ISO, Year, dln_e, pi)
}

#+begin_src R
build_gap_panel <- function() {
  get_gIO() |>
    inner_join(get_gOff(),    by = c("ISO", "Year")) |>
    inner_join(get_xr_infl(), by = c("ISO", "Year")) |>
    mutate(Gap = gY_io - gY_off)
}

## First-order classification from simple correlations of the gap.
classify_series <- function(panel) {
  ce <- cor(panel$Gap, panel$dln_e, use = "complete.obs")
  cp <- cor(panel$Gap, panel$pi,    use = "complete.obs")
  dplyr::case_when(
    ce > 0.5 & cp < 0.3 ~ "real",
    cp > 0.5 & ce < 0.3 ~ "nominal_lcu",
    ce > 0.5 & cp > 0.5 ~ "nominal_usd",
    TRUE                ~ "ambiguous"
  )
}

#+begin_src R
fit_gap_model <- function(panel) {
  m <- lm(Gap ~ dln_e + pi + factor(ISO), data = panel)
  co <- coef(m)
  ## within R2: variance explained by dln_e and pi beyond the fixed effects
  m_fe <- lm(Gap ~ factor(ISO), data = panel)
  within_r2 <- 1 - sum(resid(m)^2) / sum(resid(m_fe)^2)
  list(
    beta_e    = unname(co["dln_e"]),
    beta_pi   = unname(co["pi"]),
    within_r2 = within_r2,
    resid     = residuals(m)
  )
}

#+begin_src R
## Algebraic counterfactual: remove the year-varying exchange term from the
## IO growth, equivalent to converting at a fixed base-year rate. A full
## Stage-A re-run with a constant rate is a stronger confirmation when the
## construction pipeline becomes runnable.
counterfactual_fixed_rate <- function(panel) {
  cf <- panel |>
    mutate(Gap_fixed = Gap - dln_e) |>
    select(ISO, Year, Gap_fixed)
  cor_before <- cor(panel$Gap, panel$dln_e, use = "complete.obs")
  cor_after  <- cor(cf$Gap_fixed, panel$dln_e, use = "complete.obs")
  attr(cf, "cor_drop") <- cor_before - cor_after
  cf
}
#+end_src

## Fallback countries are those get_deflator() sources from the World Bank,
## identified as ISO present in get_deflator() but absent from the OCDE block.
attribute_residual <- function(panel, fit) {
  res <- panel |> mutate(r = fit$resid)

  ## H3: residual magnitude against horizon length within each country
  h3 <- res |>
    group_by(ISO) |>
    mutate(horizon = Year - min(Year)) |>
    ungroup() |>
    summarise(slope = coef(lm(abs(r) ~ horizon))["horizon"]) |>
    pull(slope)

  tibble(
    test      = c("H3_horizon_slope"),
    statistic = c(h3),
    reading   = c("positive slope supports chain-vs-fixed substitution bias")
  )
}

## H5 (sector aggregation) and H6 (base year) cannot move a growth rate under
## the current construction. The checks confirm the invariants numerically.
exclusion_checks <- function() {
  results <- load_io()
  iso <- names(results)[1]
  lvl <- as.numeric(results[[iso]][["GDP"]][, "GDP"])
  g_base1 <- diff(log(lvl))
  g_base2 <- diff(log(lvl * 100))     # rescale as a base-year change proxy
  tibble(
    hypothesis = c("H6_base_year"),
    invariant  = c("growth invariant to constant rescaling"),
    holds      = c(isTRUE(all.equal(g_base1, g_base2)))
  )
}

#+begin_src R
run_diagnosis <- function() {
  panel <- build_gap_panel()
  fit   <- fit_gap_model(panel)
  cls   <- classify_series(panel)
  cf    <- counterfactual_fixed_rate(panel)

  p <- ggplot(panel, aes(dln_e, Gap)) +
    geom_point(alpha = 0.4) +
    geom_smooth(method = "lm", se = FALSE) +
    labs(x = expression(Delta*ln(e)), y = "Gap = g_IO - g_off")
  ggsave("../figs/gap_vs_dln_e.pdf", p, width = 8, height = 6)

  tab <- tibble(
    Quantity = c("beta_e", "beta_pi", "within_R2", "series_class", "cor_drop"),
    Value    = c(round(fit$beta_e, 3), round(fit$beta_pi, 3),
                 round(fit$within_r2, 3), cls, round(attr(cf, "cor_drop"), 3))
  )
  writeLines(
    knitr::kable(tab, format = "latex", booktabs = TRUE,
                 caption = "Exchange-rate growth-gap diagnosis"),
    "../tabs/xr_gap_diagnosis.tex"
  )
  list(panel = panel, fit = fit, class = cls, counterfactual = cf)
}
#+end_src
